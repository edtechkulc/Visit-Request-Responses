// ===================================================================== // ⚙️ CONFIG — แก้ค่าตรงนี้ // ===================================================================== const SHEET_NAME = 'Form responses 1'; // ชื่อชีตที่มีข้อมูล (tab ล่างสุด) const CLEAR_PIN = '1234'; // PIN สำหรับยืนยันก่อนล้างข้อมูล (เปลี่ยนได้) const HEADER_ROWS = 1; // จำนวนแถว header ที่จะคงไว้ (ปกติ = 1) // ===================================================================== /** * doGet — ใช้ทดสอบว่า Web App ทำงานได้ * เปิด URL ของ Web App ในเบราว์เซอร์ จะเห็น {"status":"ok"} */ function doGet(e) { return jsonResponse({ status: 'ok', message: 'Visit Request Responses API is running.' }); } /** * doPost — รับคำสั่งจาก index.html * * Request body (JSON string, Content-Type: text/plain เพื่อหลีกเลี่ยง CORS preflight): * { "action": "clear", "pin": "1234" } * * Response JSON: * { "status": "ok", "message": "..." } ← สำเร็จ * { "status": "error", "message": "..." } ← ผิดพลาด */ function doPost(e) { try { // --- Parse body --- let body; try { const raw = e.postData ? e.postData.contents : ''; body = JSON.parse(raw); } catch (_) { return jsonResponse({ status: 'error', message: 'Invalid JSON body' }); } const action = (body.action || '').trim(); const pin = (body.pin || '').trim(); // --- Route actions --- if (action === 'clear') { return handleClear(pin); } return jsonResponse({ status: 'error', message: `Unknown action: ${action}` }); } catch (err) { return jsonResponse({ status: 'error', message: err.message }); } } // ── Clear all data rows (keep header) ────────────────────────────── function handleClear(pin) { if (pin !== CLEAR_PIN) { return jsonResponse({ status: 'error', message: 'PIN ไม่ถูกต้อง' }); } const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { return jsonResponse({ status: 'error', message: `ไม่พบชีต "${SHEET_NAME}"` }); } const lastRow = sheet.getLastRow(); if (lastRow <= HEADER_ROWS) { return jsonResponse({ status: 'ok', message: 'ไม่มีข้อมูลที่ต้องล้าง' }); } // ลบทุกแถวหลัง header const deleteFrom = HEADER_ROWS + 1; const deleteCount = lastRow - HEADER_ROWS; sheet.deleteRows(deleteFrom, deleteCount); // Log การล้างข้อมูล (ดูได้ใน Apps Script → Executions) console.log(`[${new Date().toISOString()}] Cleared ${deleteCount} rows from "${SHEET_NAME}"`); return jsonResponse({ status: 'ok', message: `ล้างข้อมูลสำเร็จ ${deleteCount} รายการ`, deletedRows: deleteCount }); } // ── Helper: return JSON ContentService response ───────────────────── function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
